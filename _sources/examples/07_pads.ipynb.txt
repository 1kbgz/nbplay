    "## Pads + Sampler via Session\n",
    "\n",
    "Connect the pads to a sampler through the session bus — the same\n",
    "pattern used by `KeyboardWidget` and `MidiKeyboardWidget`."
   ]
  },
  {
   "cell_type": "code",
   "execution_count": null,
   "id": "6",
   "metadata": {
    "id": "6",
    "language": "python"
   },
   "outputs": [],
   "source": [
    "import math\n",
    "from nbplay import SamplerWidget, PadWidget\n",
    "\n",
    "# Create a simple sine-wave \"sample\" for the sampler to play\n",
    "sample_rate = 44100\n",
    "duration = 0.5\n",
    "freq = 440.0\n",
    "frames = int(sample_rate * duration)\n",
    "samples = [math.sin(2 * math.pi * freq * i / sample_rate) * 0.6 for i in range(frames)]\n",
    "\n",
    "sampler = SamplerWidget(\n",
    "    channel_index=0,\n",
    "    root_note=60,\n",
    "    attack=0.001,\n",
    "    decay=0.05,\n",
    "    sustain=0.8,\n",
    "    release=0.1,\n",
    "    max_voices=8,\n",
    ")\n",
    "sampler.load_sample(samples, sample_rate=sample_rate, root_note=60, name=\"Sine\")\n",
    "\n",
    "# 4×4 drum-style pads connected to the sampler\n",
    "pads = PadWidget(rows=4, cols=4, velocity=110)\n",
    "pads.connect_sampler(sampler)\n",
    "\n",
    "sampler, pads\n"
   ]
